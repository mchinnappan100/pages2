5 · Cosine similarity

Measure semantic similarity between two textsbasic
import rllm
import math

def cosine_sim(a: list, b: list) -> float:
    dot   = sum(x * y for x, y in zip(a, b))
    mag_a = math.sqrt(sum(x * x for x in a))
    mag_b = math.sqrt(sum(x * x for x in b))
    return dot / (mag_a * mag_b + 1e-9)

pairs = [
    ("dog",             "puppy"),
    ("machine learning", "deep learning"),
    ("apple",           "orange"),
    ("Rust",            "Python"),
]

for a, b in pairs:
    sim = cosine_sim(rllm.embed(a), rllm.embed(b))
    print(f"{a!r:20s} ↔ {b!r:20s}  sim={sim:.4f}")

# Verified output (Python 3.14.2, GPT-2, macOS arm64):
# session.embed("neural network") ↔ session.embed("deep learning")  sim=0.4348

6 · Semantic search over a list

Find the most similar document to a queryintermediate
import rllm, math

def cosine(a, b):
    dot = sum(x*y for x,y in zip(a,b))
    return dot / (math.sqrt(sum(x*x for x in a)) * math.sqrt(sum(x*x for x in b)) + 1e-9)

docs = [
    "Rust guarantees memory safety without a garbage collector.",
    "Python is great for data science and machine learning.",
    "candle is a Rust ML framework built by Hugging Face.",
    "The rllm CLI runs LLMs locally on Metal and CUDA.",
    "GPT-2 is a transformer language model from OpenAI.",
]

query = "fast LLM inference on Apple Silicon"
q_vec = rllm.embed(query)
doc_vecs = [rllm.embed(d) for d in docs]

ranked = sorted(enumerate(docs), key=lambda i_d: cosine(q_vec, doc_vecs[i_d[0]]), reverse=True)

print(f"Query: {query}\n")
for rank, (i, doc) in enumerate(ranked, 1):
    score = cosine(q_vec, doc_vecs[i])
    print(f"  #{rank}  ({score:.4f})  {doc}")

7 · Batch embeddings with numpy

Embed a list of sentences into a 2-D numpy arrayintermediate
import rllm
import numpy as np

sentences = [
    "The model weights are stored in safetensors format.",
    "candle-core provides tensor operations on Metal.",
    "rllm infer generate runs GPT-2 inference locally.",
    "PyO3 lets you call Rust code from Python.",
]

# Embed each sentence → stack into (N, 768) matrix
matrix = np.array([rllm.embed(s) for s in sentences], dtype=np.float32)

print("shape :", matrix.shape)       # (4, 768)
print("dtype :", matrix.dtype)       # float32
print("norms :", np.linalg.norm(matrix, axis=1))

# Full pairwise cosine similarity matrix
norms = np.linalg.norm(matrix, axis=1, keepdims=True)
normed = matrix / (norms + 1e-9)
sim_matrix = normed @ normed.T
print("\nCosine similarity matrix:\n", np.round(sim_matrix, 3))

8 · Embedding matrix + nearest neighbour

Build an index then query itintermediate
import rllm
import numpy as np

class EmbeddingIndex:
    def __init__(self):
        self.texts = []
        self.matrix = None

    def add(self, texts: list[str]):
        self.texts.extend(texts)
        vecs = np.array([rllm.embed(t) for t in texts], dtype=np.float32)
        self.matrix = vecs if self.matrix is None else np.vstack([self.matrix, vecs])

    def query(self, text: str, top_k: int = 3) -> list[tuple[float, str]]:
        q = np.array(rllm.embed(text), dtype=np.float32)
        q /= (np.linalg.norm(q) + 1e-9)
        m = self.matrix / (np.linalg.norm(self.matrix, axis=1, keepdims=True) + 1e-9)
        scores = m @ q
        top = np.argsort(scores)[::-1][:top_k]
        return [(float(scores[i]), self.texts[i]) for i in top]

idx = EmbeddingIndex()
idx.add([
    "rllm runs LLMs on Metal GPU.",
    "GPT-2 uses transformer blocks with causal self-attention.",
    "maturin builds PyO3 extensions into Python wheels.",
    "rllm rag stores embeddings in an rsqlite database.",
])

for score, doc in idx.query("inference on Apple Silicon"):
    print(f"{score:.4f}  {doc}")

9 · Sentence similarity heatmap

Visualise pairwise similarity with matplotlibintermediate
import rllm
import numpy as np
import matplotlib.pyplot as plt

sentences = [
    "Rust memory safety", "C++ pointers",
    "GPT-2 language model", "BERT transformer",
    "Metal GPU compute", "CUDA kernels",
]

vecs = np.array([rllm.embed(s) for s in sentences], dtype=np.float32)
norms = np.linalg.norm(vecs, axis=1, keepdims=True)
normed = vecs / (norms + 1e-9)
sim = normed @ normed.T

fig, ax = plt.subplots(figsize=(7, 6))
im = ax.imshow(sim, cmap="YlOrRd", vmin=0, vmax=1)
ax.set_xticks(range(len(sentences))); ax.set_yticks(range(len(sentences)))
ax.set_xticklabels(sentences, rotation=45, ha="right", fontsize=8)
ax.set_yticklabels(sentences, fontsize=8)
plt.colorbar(im, ax=ax, label="Cosine similarity")
ax.set_title("rllm Embedding Similarity Heatmap")
plt.tight_layout()
plt.savefig("similarity_heatmap.png", dpi=150)
print("Saved similarity_heatmap.png")

10 · Pandas DataFrame embeddings

import rllm
import pandas as pd
import numpy as np

df = pd.DataFrame({
    "id":   [1, 2, 3, 4],
    "text": [
        "rllm is a Rust LLM CLI",
        "candle is a Rust ML framework",
        "GPT-2 runs on Metal GPU via rllm",
        "PyO3 bridges Rust and Python",
    ],
})

# Add embedding column (each cell is a 768-dim list)
df["embedding"] = df["text"].apply(rllm.embed)

# Convert to numpy matrix for downstream ML
emb_matrix = np.vstack(df["embedding"].values).astype(np.float32)

print("\nEmbedding matrix shape:", emb_matrix.shape)   # (4, 768)

# Save to parquet for later use
df["embedding_str"] = df["embedding"].apply(lambda v: ",".join(map(str, v)))
df[["id", "text", "embedding_str"]].to_parquet("embeddings.parquet", index=False)
print("Saved embeddings.parquet")

11 · Simple RAG pipeline

Retrieve relevant context then generate a grounded answeradvanced
import rllm, math

KNOWLEDGE_BASE = [
    "rllm is a pure-Rust LLM CLI with 25 crates across 5 phases.",
    "rllm infer generate runs GPT-2 inference on Metal GPU.",
    "rllm rag stores embeddings in an rsqlite database with cosine similarity.",
    "rllm mcp serves an OpenAI-compatible API and an MCP protocol server.",
    "rllm agent runs a ReAct tool-calling loop with built-in search, calc, file, shell tools.",
    "rllm pyo3 exposes Rust inference to Python via PyO3 abi3 wheels.",
]

def cosine(a, b):
    dot = sum(x*y for x,y in zip(a,b))
    return dot / (math.sqrt(sum(x*x for x in a))*math.sqrt(sum(x*x for x in b)) + 1e-9)

def retrieve(query: str, top_k: int = 2) -> list[str]:
    q_vec = rllm.embed(query)
    scored = [(cosine(q_vec, rllm.embed(doc)), doc) for doc in KNOWLEDGE_BASE]
    return [d for _, d in sorted(scored, reverse=True)[:top_k]]

def rag_answer(question: str) -> str:
    context = retrieve(question)
    prompt = f"Context:\n{chr(10).join(f'- {c}' for c in context)}\n\nQ: {question}\nA:"
    return rllm.generate(prompt, max_tokens=60)

questions = [
    "How do I run inference with rllm?",
    "What tools does the rllm agent support?",
    "How can I use rllm from Python?",
]

for q in questions:
    print(f"Q: {q}")
    print(f"A: {rag_answer(q)}\n")

12 · Flask embedding API

Serve rllm embeddings over HTTP with Flaskadvanced
# pip install flask
from flask import Flask, request, jsonify
import rllm

app = Flask(__name__)

@app.route("/embed", methods=["POST"])
def embed_endpoint():
    data = request.get_json()
    text = data.get("text", "")
    if not text:
        return jsonify({"error": "missing text"}), 400
    vec = rllm.embed(text)
    return jsonify({"text": text, "embedding": vec, "dim": len(vec)})

@app.route("/generate", methods=["POST"])
def generate_endpoint():
    data = request.get_json()
    prompt = data.get("prompt", "")
    max_tokens = data.get("max_tokens", 20)
    return jsonify({"output": rllm.generate(prompt, max_tokens=max_tokens)})

@app.route("/version")
def version_endpoint():
    return jsonify({"version": rllm.version()})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5050, debug=False)

# curl -s -X POST http://localhost:5050/embed \
#      -H "Content-Type: application/json" \

14 · Async generation wrapper

Wrap the synchronous binding in asyncio for use in async applicationsadvanced
import asyncio, rllm
from concurrent.futures import ThreadPoolExecutor

_executor = ThreadPoolExecutor(max_workers=4)

async def async_generate(prompt: str, max_tokens: int = 20) -> str:
    loop = asyncio.get_running_loop()
    return await loop.run_in_executor(_executor, rllm.generate, prompt, "gpt2", max_tokens)

async def async_embed(text: str) -> list[float]:
    loop = asyncio.get_running_loop()
    return await loop.run_in_executor(_executor, rllm.embed, text)

async def main():
    prompts = ["The sky is", "Rust's ownership", "In 2026"]
    # Run all three generations concurrently
    results = await asyncio.gather(*[async_generate(p, 25) for p in prompts])
    for prompt, result in zip(prompts, results):
        print(f"{prompt!r:25s} → {result}")

asyncio.run(main())

15 · CLI tool using rllm bindings

Build a small command-line tool on top of the Python bindingsadvanced
#!/usr/bin/env python3
"""rllm-py — tiny CLI wrapping the rllm Python bindings.
Usage:
  rllm-py generate "The quick brown"
  rllm-py embed "hello world" --dim 10
  rllm-py similarity "cat" "kitten"
  rllm-py search "fast inference" doc1.txt doc2.txt doc3.txt
"""
import sys, math, rllm

def cosine(a, b):
    dot = sum(x*y for x,y in zip(a,b))
    return dot / (math.sqrt(sum(x*x for x in a))*math.sqrt(sum(x*x for x in b))+1e-9)

def cmd_generate(args):
    prompt = " ".join(args) or "Hello"
    print(rllm.generate(prompt, max_tokens=40))

def cmd_embed(args):
    dim = None
    if "--dim" in args:
        i = args.index("--dim"); dim = int(args[i+1]); args = args[:i] + args[i+2:]
    text = " ".join(args)
    vec = rllm.embed(text)
    print(f"dim={len(vec)}", vec[:dim] if dim else vec[:5], "...")

def cmd_similarity(args):
    a, b = args[0], args[1]
    sim = cosine(rllm.embed(a), rllm.embed(b))
    print(f"{a!r} ↔ {b!r}  cosine={sim:.4f}")

def cmd_search(args):
    query, files = args[0], args[1:]
    q_vec = rllm.embed(query)
    results = []
    for f in files:
        text = open(f).read()[:500]
        results.append((cosine(q_vec, rllm.embed(text)), f))
    for score, f in sorted(results, reverse=True):
        print(f"{score:.4f}  {f}")

CMDS = {"generate": cmd_generate, "embed": cmd_embed,
        "similarity": cmd_similarity, "search": cmd_search}

if __name__ == "__main__":
    args = sys.argv[1:]
    cmd = args[0] if args else "generate"
    CMDS.get(cmd, cmd_generate)(args[1:])
⚠ REPL note: save to rllm-py.py and run with python3 rllm-py.py generate "hello".

16 · Session: load once, generate many

rllm.load() — load model into memory once, reuse itintermediate
import rllm

# Load once — model weights stay in memory for the life of `session`.
# Requires a local model directory (config.json + model.safetensors + tokenizer.json)
session = rllm.load("~/.rllm/models/gpt2")

print(session)                 # RllmSession(path='~/.rllm/models/gpt2')
print(session.model_id)        # gpt2

prompts = [
    "The quick brown fox",
    "In the beginning",
    "Rust is fast because",
]

for p in prompts:
    out = session.generate(p, max_tokens=25)
    print(f"{p!r:35s} → {out}")

# All three prompts use the same in-memory model — no disk I/O after the first load.

17 · Session: sampling parameters

Control temperature, top-k, top-p, seedintermediate
import rllm

session = rllm.load("~/.rllm/models/gpt2")
prompt  = "Once upon a time in a land far away"

# Greedy — deterministic, picks the highest-probability token every step
greedy = session.generate(prompt, max_tokens=30, temperature=0.0)
print("greedy :", greedy)

# Creative — high temperature, top-p nucleus sampling
creative = session.generate(prompt, max_tokens=30, temperature=1.2, top_p=0.95)
print("creative:", creative)

# Reproducible — fix the RNG seed
run1 = session.generate(prompt, max_tokens=20, temperature=0.8, seed=42)
run2 = session.generate(prompt, max_tokens=20, temperature=0.8, seed=42)
print("seed=42, run 1:", run1)
print("seed=42, run 2:", run2)
assert run1 == run2, "same seed → same output"
print("✓ reproducible")

# Top-k only — restricts vocabulary to top 40 tokens at each step
topk = session.generate(prompt, max_tokens=30, temperature=0.9, top_k=40)
print("top-k=40:", topk)

18 · Session: Metal GPU + reuse across calls

rllm.load_metal() for Apple Silicon, then share the sessionadvanced
import rllm, time

# load_metal() uses Apple Metal GPU; falls back to CPU if Metal unavailable.
# On M-series Macs this gives 3-5× faster inference vs CPU.
session = rllm.load_metal("~/.rllm/models/gpt2")
print("model:", session.model_id)

# Share a single session across a module — no reload on each call
class TextGenerator:
    def __init__(self, session: rllm.RllmSession):
        self._session = session

    def complete(self, prompt: str, max_tokens: int = 40) -> str:
        return self._session.generate(prompt, max_tokens=max_tokens, temperature=0.85, top_p=0.92)

    def batch(self, prompts: list[str], max_tokens: int = 40) -> list[str]:
        return [self.complete(p, max_tokens) for p in prompts]

gen = TextGenerator(session)

start = time.perf_counter()
results = gen.batch([
    "The transformer architecture",
    "Apple Silicon is powerful because",
    "Rust's ownership system ensures",
    "In 2026, LLMs can",
], max_tokens=35)
elapsed = time.perf_counter() - start

for prompt, out in zip(["The transformer","Apple Silicon","Rust ownership","In 2026"], results):
    print(f"{prompt:20s} → {out[:60]}...")
print(f"\n4 prompts in {elapsed:.2f}s  ({elapsed/4*1000:.0f} ms/prompt avg)")
