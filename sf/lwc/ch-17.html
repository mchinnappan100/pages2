Community Context

Detecting Experience Cloud Context

import { LightningElement, wire } from 'lwc';
import { CurrentPageReference } from 'lightning/navigation';
import communityId from '@salesforce/community/Id';
import communityBasePath from '@salesforce/community/basePath';
import isGuestUser from '@salesforce/user/isGuest';

export default class CommunityAware extends LightningElement {
    // Community-specific imports
    get communityId() { return communityId; }
    get communityPath() { return communityBasePath; }
    get isGuest() { return isGuestUser; }

    // Current page reference includes community context
    @wire(CurrentPageReference) pageRef;
    get communityName() { return this.pageRef?.state?.communityName; }

    // Conditional rendering for guest vs authenticated
    get showLoginPrompt() { return this.isGuest; }
    get showContent()     { return !this.isGuest; }
}

targetConfigs — Community Builder

Configurable Properties in Experience Builder

// js-meta.xml with targetConfigs for Experience Cloud
// <targets>
//   <target>lightningCommunity__Page</target>
//   <target>lightningCommunity__Default</target>
// </targets>
// <targetConfigs>
//   <targetConfig targets="lightningCommunity__Page">
//     <property name="pageTitle" type="String" label="Page Title"
//               default="Welcome" description="Title shown on the page">
//     </property>
//     <property name="showHero" type="Boolean" label="Show Hero Banner"
//               default="true">
//     </property>
//     <property name="maxItems" type="Integer" label="Max Items"
//               min="1" max="50" default="10">
//     </property>
//   </targetConfig>
// </targetConfigs>

// In JS — these become @api properties automatically:
import { LightningElement, api } from 'lwc';
export default class HeroSection extends LightningElement {
    @api pageTitle  = 'Welcome';
    @api showHero   = true;
    @api maxItems   = 10;
    // Admin sets these via Experience Builder drag-and-drop
}

Design Attributes — Branding

Community Theme Design Tokens

// Experience Cloud applies community brand colours as CSS variables
// Your component automatically picks them up via SLDS tokens

// component CSS:
:host {
    background-color: var(--dxp-g-brand);        // primary brand colour
    color: var(--dxp-g-brand-contrast-high);     // text on brand background
    border-color: var(--dxp-g-brand-secondary);  // secondary brand
    border-radius: var(--dxp-c-button-radius-border);
}
.button {
    background: var(--dxp-g-root-action);
    color: var(--dxp-g-root-action-contrast);
    font-family: var(--dxp-g-font-family-brand);
}

// Community admin can change these without code changes
// via Experience Builder → Branding panel

Community Navigation

NavigationMixin in Experience Cloud

import { NavigationMixin } from 'lightning/navigation';
import { LightningElement } from 'lwc';
import communityBasePath from '@salesforce/community/basePath';

export default class CommunityNav extends NavigationMixin(LightningElement) {

    // Navigate to a community page
    navigateToContactUs() {
        this[NavigationMixin.Navigate]({
            type: 'comm__namedPage',
            attributes: { name: 'Contact_Us__c' }
        });
    }

    // Navigate to a community record detail page
    navigateToRecord(recordId) {
        this[NavigationMixin.Navigate]({
            type: 'standard__recordPage',
            attributes: { recordId, actionName: 'view' }
        });
    }

    // Build a full URL for an anchor tag
    connectedCallback() {
        this[NavigationMixin.GenerateUrl]({
            type: 'comm__namedPage',
            attributes: { name: 'Home' }
        }).then(url => { this.homeUrl = url; });
    }
}
05

Best Practices

Always Check isGuestUser Before Showing Data

Guest users have limited data access. Always check isGuestUser and show appropriate prompts rather than blank screens or access errors.

✅ <template lwc:if={isGuest}>Please log in...</template><template lwc:else>{content}</template>

Use dxp Design Tokens for Branding

Hard-coded colours ignore the community brand theme. Use var(--dxp-g-brand) and related tokens so your component adapts to any community's branding.

❌ background: #1b5297; /* ignores community theme */
✅ background: var(--dxp-g-brand);

Define targetConfigs for All Community Properties

Any admin-configurable property must be declared in targetConfigs in js-meta.xml. Without this, properties won't appear in Experience Builder.

✅ <targetConfig targets="lightningCommunity__Page"><property name="title"...>
