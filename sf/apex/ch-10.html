@RestResource — Apex REST Endpoints

Annotating a class with @RestResource exposes it as a custom REST endpoint accessible at /services/apexrest/YourPath.

Basic REST Resource Structure

// URL: /services/apexrest/accounts/
@RestResource(urlMapping='/accounts/*')
global with sharing class AccountRestService {

    // GET /services/apexrest/accounts/{id}
    @HttpGet
    global static AccountDTO doGet() {
        RestRequest req  = RestContext.request;
        RestResponse res = RestContext.response;

        // Extract the Id from the URL path
        String accountId = req.requestURI.substring(
            req.requestURI.lastIndexOf('/') + 1);

        if (String.isBlank(accountId)) {
            res.statusCode = 400;
            return null;
        }

        List<Account> accs =
            [SELECT Id, Name, Industry, AnnualRevenue
             FROM Account WHERE Id = :accountId
             WITH SECURITY_ENFORCED LIMIT 1];

        if (accs.isEmpty()) {
            res.statusCode = 404;
            return null;
        }

        res.statusCode = 200;
        return new AccountDTO(accs[0]);
    }

    // DTO — data transfer object for clean serialisation
    global class AccountDTO {
        global String id;
        global String name;
        global String industry;
        global Decimal annualRevenue;

        global AccountDTO(Account a) {
            this.id            = a.Id;
            this.name          = a.Name;
            this.industry      = a.Industry;
            this.annualRevenue = a.AnnualRevenue;
        }
    }
}

HTTP Method Annotations

📥

@HttpGet — Read

Retrieve resources. Must be idempotent — no side effects.

  • No request body
  • Returns serialised response
  • URL parameters via req.params
  • Should return 200 on success
  • 404 when resource not found
  • Use WITH SECURITY_ENFORCED
@HttpGet
global static List<Account> doGet() {
    RestRequest req = RestContext.request;
    String industry = req.params.get('industry');
    String limitStr = req.params.get('limit');
    Integer maxRows = limitStr != null ? Integer.valueOf(limitStr) : 10;

    // ?industry=Technology&limit=20
    List<Account> results;
    if (String.isNotBlank(industry)) {
        results = [SELECT Id, Name, Industry, AnnualRevenue
                   FROM Account
                   WHERE Industry = :industry
                   WITH SECURITY_ENFORCED
                   ORDER BY AnnualRevenue DESC NULLS LAST
                   LIMIT :maxRows];
    } else {
        results = [SELECT Id, Name, Industry
                   FROM Account
                   WITH SECURITY_ENFORCED
                   LIMIT :maxRows];
    }

    RestContext.response.statusCode = 200;
    return results;
}
📤

@HttpPost — Create

Create a new resource. Request body is auto-deserialised into the method parameter.

  • Request body auto-deserialised
  • Return 201 Created on success
  • Return the created resource
  • Validate required fields
  • Catch DmlException
  • Include Location header
@HttpPost
global static CreateResult doPost(
        String name, String industry, Decimal annualRevenue) {
    RestResponse res = RestContext.response;

    if (String.isBlank(name)) {
        res.statusCode = 400;
        return new CreateResult(false, null, 'Name is required');
    }

    Account acc = new Account(
        Name          = name,
        Industry      = industry,
        AnnualRevenue = annualRevenue
    );

    try {
        insert acc;
        res.statusCode = 201;
        // Standard REST practice: return Location header
        res.headers.put('Location',
            '/services/apexrest/accounts/' + acc.Id);
        return new CreateResult(true, acc.Id, null);
    } catch (DmlException e) {
        res.statusCode = 422;
        return new CreateResult(false, null, e.getDmlMessage(0));
    }
}

global class CreateResult {
    global Boolean success;
    global String  id;
    global String  error;
    global CreateResult(Boolean s, String i, String e) {
        success=s; id=i; error=e;
    }
}
✏️

@HttpPatch / @HttpPut / @HttpDelete

Update and delete operations with appropriate HTTP semantics.

  • PATCH: partial update (only changed fields)
  • PUT: full replacement of resource
  • DELETE: removes resource
  • Return 200 for PATCH/PUT
  • Return 204 No Content for DELETE
  • Return 404 if resource not found
// PATCH — partial update
@HttpPatch
global static UpdateResult doPatch(
        String industry, Decimal annualRevenue) {
    RestRequest req = RestContext.request;
    RestResponse res = RestContext.response;

    String accountId = req.requestURI.substring(
        req.requestURI.lastIndexOf('/') + 1);

    List<Account> accs =
        [SELECT Id, Industry, AnnualRevenue FROM Account
         WHERE Id = :accountId WITH SECURITY_ENFORCED LIMIT 1];

    if (accs.isEmpty()) {
        res.statusCode = 404;
        return new UpdateResult(false, 'Account not found');
    }

    Account acc = accs[0];
    if (industry      != null) acc.Industry      = industry;
    if (annualRevenue != null) acc.AnnualRevenue = annualRevenue;

    update acc;
    res.statusCode = 200;
    return new UpdateResult(true, null);
}

// DELETE
@HttpDelete
global static void doDelete() {
    RestRequest req = RestContext.request;
    RestResponse res = RestContext.response;

    String accountId = req.requestURI.substring(
        req.requestURI.lastIndexOf('/') + 1);

    List<Account> accs =
        [SELECT Id FROM Account WHERE Id = :accountId
         WITH SECURITY_ENFORCED LIMIT 1];

    if (accs.isEmpty()) { res.statusCode = 404; return; }
    delete accs[0];
    res.statusCode = 204;
}

global class UpdateResult {
    global Boolean success; global String error;
    global UpdateResult(Boolean s, String e) { success=s; error=e; }
}

RestContext — Request & Response

RestContext gives access to the full HTTP request and lets you control the response precisely.

  • RestContext.request — incoming request
  • RestContext.response — outgoing response
  • req.requestURI — URL path
  • req.params — query string map
  • req.headers — request headers
  • req.requestBody — raw Blob body
@HttpPost
global static void doPost() {
    RestRequest  req = RestContext.request;
    RestResponse res = RestContext.response;

    // Read headers
    String contentType = req.headers.get('Content-Type');
    String apiKey      = req.headers.get('X-API-Key');

    // Read raw body
    String bodyStr = req.requestBody.toString();

    // Deserialise JSON body
    Map<String, Object> payload =
        (Map<String, Object>) JSON.deserializeUntyped(bodyStr);

    // Get URL parameter: /accounts?source=mobile
    String source = req.params.get('source');

    // Extract ID from URL path: /accounts/001xxx
    String[] parts = req.requestURI.split('/');
    String recordId = parts[parts.size() - 1];

    // Set response
    res.statusCode = 201;
    res.headers.put('Content-Type', 'application/json');
    res.headers.put('X-Request-Id', EncodingUtil.convertToHex(
        Crypto.generateAesKey(128)));
    res.responseBody = Blob.valueOf(
        JSON.serialize(new Map<String, String>{
            'status' => 'created',
            'id'     => recordId
        })
    );
}

JSON Serialisation & DTOs

Controlling JSON Output with @AuraEnabled / Wrapper DTOs

// Custom DTO controls exactly what JSON is returned
// Avoids exposing internal field names or sensitive data
global class AccountListResponse {
    global Integer     total;
    global String      nextPageToken;
    global List<AccountSummary> records;

    global AccountListResponse(List<Account> accs, Integer total) {
        this.total   = total;
        this.records = new List<AccountSummary>();
        for (Account a : accs) {
            this.records.add(new AccountSummary(a));
        }
    }
}

global class AccountSummary {
    global String  id;
    global String  name;
    global String  industry;
    global Decimal revenue;  // renamed from AnnualRevenue
    global String  type;

    global AccountSummary(Account a) {
        this.id       = a.Id;
        this.name     = a.Name;
        this.industry = a.Industry;
        this.revenue  = a.AnnualRevenue;
        this.type     = a.Type;
    }
}

// Returns clean JSON:
// {
//   "total": 150,
//   "nextPageToken": null,
//   "records": [{ "id": "...", "name": "ACME", "revenue": 500000 }]
// }

Apex SOAP Web Services

Use webservice keyword to expose methods as SOAP operations. Salesforce auto-generates the WSDL.

Defining a SOAP Web Service

// global class + webservice keyword = SOAP endpoint
global with sharing class AccountWebService {

    // webservice methods must be static and in a global class
    webservice static String createAccount(
            String name, String industry) {

        if (String.isBlank(name)) {
            throw new WebServiceException('Name is required');
        }

        Account acc = new Account(
            Name     = name,
            Industry = industry
        );
        insert acc;
        return acc.Id;
    }

    webservice static AccountInfo getAccount(Id accountId) {
        List<Account> accs =
            [SELECT Id, Name, Industry, AnnualRevenue
             FROM Account WHERE Id = :accountId LIMIT 1];
        if (accs.isEmpty()) return null;
        return new AccountInfo(accs[0]);
    }

    // Inner class types are included in the WSDL
    global class AccountInfo {
        webservice String  id;
        webservice String  name;
        webservice String  industry;
        webservice Decimal annualRevenue;

        global AccountInfo(Account a) {
            this.id            = a.Id;
            this.name          = a.Name;
            this.industry      = a.Industry;
            this.annualRevenue = a.AnnualRevenue;
        }
    }
}

// Access WSDL at:
// https://yourinstance.salesforce.com/services/wsdl/class/AccountWebService
REST vs SOAP: Prefer REST for new integrations — it's simpler, more widely supported, and requires no WSDL management. Use SOAP only when integrating with legacy enterprise systems that require it.

REST API Security

Authentication Methods

  • OAuth 2.0 (recommended): Connected App + JWT or User-Agent flow
  • Session ID: Pass Authorization: Bearer {sessionId}
  • Named Principal / Per-User: OAuth with stored credentials
  • Never accept unauthenticated requests to Apex REST
  • Use IP restrictions on the Connected App

Authorization in Apex REST

  • Always use with sharing — enforces record-level access
  • Use WITH SECURITY_ENFORCED in SOQL
  • Check Schema.sObjectType.isAccessible() for CRUD
  • Validate all input — treat external callers as untrusted
  • Rate-limit via Connected App policies

Best Practices

Always Set statusCode Explicitly

Apex REST defaults to 200 but you should always set it to communicate intent: 201 Created, 204 No Content, 400 Bad Request, 404 Not Found, 422 Unprocessable Entity.

❌ // No statusCode set — caller gets 200 even on errors
✅ RestContext.response.statusCode = 404;

Use DTOs, Not Raw sObjects

Returning sObjects exposes all queried fields and internal field names. DTOs let you rename fields, filter sensitive data, and version your API independently.

❌ return [SELECT Id, SSN__c FROM Contact];
✅ return new ContactDTO(contact); // controlled output

Validate All Input

External callers are untrusted. Validate required fields, formats (Id validity, date ranges), and lengths before performing any DML.

❌ insert new Account(Name = payload.get('name'));
✅ if (String.isBlank(name)) { res.statusCode=400; return; }

Use urlMapping Wildcards Carefully

urlMapping='/accounts/*' catches all paths under /accounts/. A wildcard that is too broad can accidentally match unintended paths.

❌ urlMapping='/*' // matches everything
✅ urlMapping='/v1/accounts/*' // versioned and scoped

Version Your API in the URL

Prefix your URL with a version: /v1/accounts. When you need breaking changes, add /v2/accounts — existing consumers continue to work on v1.

❌ urlMapping='/accounts/*'
✅ urlMapping='/v1/accounts/*'

Test with RestContext Mocking

In tests, set up RestContext.request and RestContext.response manually before calling your REST methods.

✅ RestContext.request = new RestRequest();
RestContext.response = new RestResponse();
