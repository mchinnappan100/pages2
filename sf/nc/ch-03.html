HTTP Methods — GET, POST, PUT, PATCH, DELETE

Full REST API Interaction Pattern

public with sharing class RestApiService {

    private static final String NC = 'callout:My_REST_API';

    // GET — read a resource
    public static Map<String,Object> getAccount(String accountId) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + '/accounts/' + accountId);
        req.setMethod('GET');
        req.setHeader('Accept', 'application/json');
        return sendAndParse(req);
    }

    // POST — create a resource
    public static String createAccount(String name, String industry) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + '/accounts');
        req.setMethod('POST');
        req.setHeader('Content-Type', 'application/json');
        req.setBody(JSON.serialize(new Map<String,Object>{
            'name' => name, 'industry' => industry
        }));
        Map<String,Object> res = sendAndParse(req);
        return (String) res.get('id');
    }

    // PUT — full replacement
    public static void updateAccount(String id, Map<String,Object> fields) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + '/accounts/' + id);
        req.setMethod('PUT');
        req.setHeader('Content-Type', 'application/json');
        req.setBody(JSON.serialize(fields));
        send(req, 200);
    }

    // PATCH — partial update
    public static void patchAccount(String id, Map<String,Object> changes) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + '/accounts/' + id);
        req.setMethod('PATCH');
        req.setHeader('Content-Type', 'application/json');
        req.setBody(JSON.serialize(changes));
        send(req, 200);
    }

    // DELETE — remove resource
    public static void deleteAccount(String id) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + '/accounts/' + id);
        req.setMethod('DELETE');
        send(req, 204); // 204 No Content
    }

    // ─── Shared helpers ────────────────────────────────────────────────────
    private static Map<String,Object> sendAndParse(HttpRequest req) {
        HttpResponse res = new Http().send(req);
        if (res.getStatusCode() >= 400) {
            throw new CalloutException(
                'HTTP ' + res.getStatusCode() + ': ' + res.getBody());
        }
        return (Map<String,Object>) JSON.deserializeUntyped(res.getBody());
    }

    private static void send(HttpRequest req, Integer expectedStatus) {
        HttpResponse res = new Http().send(req);
        if (res.getStatusCode() != expectedStatus) {
            throw new CalloutException(
                'Expected ' + expectedStatus + ', got ' + res.getStatusCode()
                + ': ' + res.getBody());
        }
    }
}

Error Handling & Retry Patterns

Robust Error Handling

public with sharing class RobustCalloutService {

    private static final String NC = 'callout:External_API';
    private static final Integer MAX_RETRIES = 3;
    private static final Integer TIMEOUT_MS  = 10000;

    public static String callWithRetry(String path, String method, String body) {
        Integer attempts = 0;
        while (attempts < MAX_RETRIES) {
            try {
                HttpRequest req = buildRequest(path, method, body);
                HttpResponse res = new Http().send(req);

                // 2xx — success
                if (res.getStatusCode() >= 200 && res.getStatusCode() < 300) {
                    return res.getBody();
                }

                // 429 Too Many Requests — back off and retry
                if (res.getStatusCode() == 429) {
                    Integer retryAfter = Integer.valueOf(
                        res.getHeader('Retry-After') ?? '2');
                    // Note: can't sleep in Apex synchronously —
                    // use Platform Events or schedule for real back-off
                    attempts++;
                    continue;
                }

                // 401 Unauthorized — NC should auto-refresh; if still 401, give up
                if (res.getStatusCode() == 401) {
                    throw new CalloutException('Unauthorized. Check Named Credential auth config.');
                }

                // 5xx Server errors — retry
                if (res.getStatusCode() >= 500) {
                    attempts++;
                    if (attempts < MAX_RETRIES) continue;
                    throw new CalloutException('Server error after ' + MAX_RETRIES + ' retries: ' + res.getStatusCode());
                }

                // 4xx Client errors — don't retry
                Map<String,Object> err =
                    (Map<String,Object>) JSON.deserializeUntyped(res.getBody());
                throw new CalloutException(
                    'Client error ' + res.getStatusCode() + ': '
                    + (String) err.get('message'));

            } catch (CalloutException e) {
                throw e; // re-throw — don't retry on CalloutException
            } catch (System.NetException e) {
                // Network timeout — retry
                attempts++;
                if (attempts >= MAX_RETRIES) throw new CalloutException(
                    'Network error after ' + MAX_RETRIES + ' retries: ' + e.getMessage());
            }
        }
        throw new CalloutException('Max retries exceeded');
    }

    private static HttpRequest buildRequest(String path, String method, String body) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NC + path);
        req.setMethod(method);
        req.setTimeout(TIMEOUT_MS);
        req.setHeader('Content-Type', 'application/json');
        req.setHeader('Accept', 'application/json');
        if (String.isNotBlank(body)) req.setBody(body);
        return req;
    }
}

Async Callouts — @future & Queueable

Making Callouts from Triggers

// Triggers cannot make callouts directly (DML + callout in same tx = error)
// Use @future(callout=true) or Queueable + Database.AllowsCallouts

// Option 1: @future (simple, fire-and-forget)
public with sharing class ExternalSyncService {

    @future(callout=true)
    public static void syncAccountAsync(List<Id> accountIds) {
        List<Account> accounts =
            [SELECT Id, Name, Phone, Industry FROM Account WHERE Id IN :accountIds];
        String payload = JSON.serialize(accounts);

        HttpRequest req = new HttpRequest();
        req.setEndpoint('callout:CRM_Sync_API/sync/accounts');
        req.setMethod('POST');
        req.setHeader('Content-Type', 'application/json');
        req.setBody(payload);

        HttpResponse res = new Http().send(req);
        if (res.getStatusCode() != 202) {
            System.debug(LoggingLevel.ERROR, 'Sync failed: ' + res.getBody());
        }
    }
}

// Option 2: Queueable (can be chained, accepts complex types)
public with sharing class AccountSyncJob
        implements Queueable, Database.AllowsCallouts {
    private List<Account> accounts;

    public AccountSyncJob(List<Account> accounts) {
        this.accounts = accounts;
    }

    public void execute(QueueableContext ctx) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint('callout:CRM_Sync_API/sync/accounts');
        req.setMethod('POST');
        req.setHeader('Content-Type', 'application/json');
        req.setBody(JSON.serialize(accounts));
        new Http().send(req);
    }
}

// From trigger:
trigger AccountTrigger on Account (after insert, after update) {
    // Option 1:
    ExternalSyncService.syncAccountAsync(Trigger.newMap.keySet());
    // Option 2:
    System.enqueueJob(new AccountSyncJob(Trigger.new));
}

Testing Callouts with HttpCalloutMock

Mock Implementations for Unit Tests

// Mock — simulates the HTTP response
@IsTest
global class SuccessfulApiMock implements HttpCalloutMock {
    global HttpResponse respond(HttpRequest req) {
        // Verify the request was built correctly
        System.assert(req.getEndpoint().startsWith('callout:My_REST_API'),
            'Should use Named Credential endpoint');
        System.assertEquals('GET', req.getMethod());

        HttpResponse res = new HttpResponse();
        res.setStatusCode(200);
        res.setHeader('Content-Type', 'application/json');
        res.setBody('{"id":"001","name":"ACME Corp","industry":"Technology"}');
        return res;
    }
}

@IsTest
global class ErrorApiMock implements HttpCalloutMock {
    global HttpResponse respond(HttpRequest req) {
        HttpResponse res = new HttpResponse();
        res.setStatusCode(404);
        res.setBody('{"error":"Not found","message":"Account does not exist"}');
        return res;
    }
}

// Unit test
@IsTest
private class RestApiServiceTest {

    @IsTest
    static void testGetAccount_success() {
        Test.setMock(HttpCalloutMock.class, new SuccessfulApiMock());
        Test.startTest();
        Map<String,Object> result = RestApiService.getAccount('001');
        Test.stopTest();
        System.assertEquals('ACME Corp', (String) result.get('name'));
    }

    @IsTest
    static void testGetAccount_notFound() {
        Test.setMock(HttpCalloutMock.class, new ErrorApiMock());
        Test.startTest();
        try {
            RestApiService.getAccount('999');
            System.assert(false, 'Should have thrown');
        } catch (CalloutException e) {
            System.assert(e.getMessage().contains('404'));
        }
        Test.stopTest();
    }
}

Best Practices

Always Set a Timeout

Default callout timeout is 10 seconds. Always set req.setTimeout(ms) explicitly so slow external APIs don't hold your transaction hostage.

✅ req.setTimeout(10000); // fail fast after 10 seconds

Don't Retry Client Errors (4xx)

4xx errors indicate a problem with your request — retrying won't help. Only retry transient server errors (5xx) and network timeouts.

❌ if (status >= 400) retry; // retries 400 Bad Request forever
✅ if (status >= 500 || isTimeout) { retry; } else if (status >= 400) { throw; }

Use @future for Trigger Callouts

Triggers that have performed DML cannot make callouts in the same transaction. Dispatch to @future(callout=true) or Queueable + AllowsCallouts.

✅ ExternalSyncService.syncAsync(Trigger.newMap.keySet());

Log Error Responses, Not Request Bodies

Request bodies may contain PII or sensitive data. Log only error status codes and non-sensitive error messages. Never log request headers containing auth tokens.

❌ System.debug('Request: ' + req.getBody()); // may contain PII
✅ System.debug(LoggingLevel.ERROR, 'API error ' + res.getStatusCode());
