Section 1

Embedding Geometry

An embedding maps discrete tokens or documents into a continuous vector space $\mathbb{R}^d$. Proximity in this space encodes semantic relatedness. Three standard distance/similarity metrics are used depending on whether vectors are unit-normalized.

L2 distance and cosine similarity:

$$d(\mathbf{u},\mathbf{v})=\|\mathbf{u}-\mathbf{v}\|_2, \qquad \cos\theta=\frac{\mathbf{u}\cdot\mathbf{v}}{\|\mathbf{u}\|\|\mathbf{v}\|}$$

For unit-normalized vectors $\|\mathbf{u}\|=\|\mathbf{v}\|=1$, the inner product $\mathbf{u}\cdot\mathbf{v}$ equals $\cos\theta$, so dot-product retrieval is equivalent to cosine search.

import torch, torch.nn.functional as F

u = torch.randn(8, 128)   # 8 query embeddings, dim=128
v = torch.randn(32, 128)  # 32 document embeddings

# Pairwise L2 distances  (8 x 32)
dists = torch.cdist(u, v, p=2)

# Cosine similarity for first query vs all docs
cos = F.cosine_similarity(u[0].unsqueeze(0), v)  # (32,)

# Unit-normalize for dot-product retrieval
u_n = F.normalize(u, p=2, dim=-1)
v_n = F.normalize(v, p=2, dim=-1)
scores = u_n @ v_n.T   # (8 x 32) — equals cosine similarity
print(scores.shape)    # torch.Size([8, 32])
Section 2

Contrastive Learning & InfoNCE

InfoNCE (noise-contrastive estimation) trains an encoder by maximising the similarity of a query $\mathbf{q}$ with its positive key $\mathbf{k}^+$ relative to $N$ negatives $\mathbf{k}_i^-$. Temperature $\tau$ controls the sharpness of the distribution; smaller $\tau$ creates harder gradients that pull positives very tightly together.

$$\mathcal{L}=-\log\frac{\exp(\mathbf{q}\cdot\mathbf{k}^+/\tau)}{\exp(\mathbf{q}\cdot\mathbf{k}^+/\tau)+\displaystyle\sum_{i=1}^N\exp(\mathbf{q}\cdot\mathbf{k}_i^-/\tau)}$$ 
import torch, torch.nn.functional as F

def infonce_loss(q, k_pos, k_neg, tau=0.07):
    """q: (B,d), k_pos: (B,d), k_neg: (B,N,d)"""
    q  = F.normalize(q,     p=2, dim=-1)
    kp = F.normalize(k_pos, p=2, dim=-1)
    kn = F.normalize(k_neg, p=2, dim=-1)

    pos = (q * kp).sum(-1, keepdim=True) / tau        # (B,1)
    neg = torch.einsum('bd,bnd->bn', q, kn) / tau     # (B,N)

    logits = torch.cat([pos, neg], dim=-1)             # (B,1+N)
    labels = torch.zeros(logits.size(0),
                         dtype=torch.long,
                         device=q.device)              # index 0 = positive
    return F.cross_entropy(logits, labels)

loss = infonce_loss(torch.randn(16,128),
                    torch.randn(16,128),
                    torch.randn(16,31,128))
print(loss.item())
Section 3

k-Nearest Neighbour Search

Exact k-NN retrieves the $k$ closest vectors in $\mathbb{R}^d$ by exhaustive comparison. At $N=100\text{M}$ vectors with $d=768$, each query requires ~77B multiply-adds — far too slow for real-time systems. Approximate methods trade a small recall loss for orders-of-magnitude speedup.

$$\text{exact k-NN}(\mathbf{q})=\underset{i\in[N]}{\operatorname{argmin}}\;\|\mathbf{q}-\mathbf{x}_i\|_2 \qquad O(Nd)\text{ per query}$$ 
import torch

def brute_force_knn(query, corpus, k=5):
    """query: (Q,d), corpus: (N,d) -> indices (Q,k)"""
    # cdist computes pairwise L2 distances: (Q,N)
    dists = torch.cdist(query, corpus, p=2)
    topk_vals, topk_idx = torch.topk(dists, k=k,
                                      dim=-1, largest=False)
    return topk_idx, topk_vals

Q = torch.randn(10, 128)   # 10 queries
X = torch.randn(1000, 128) # 1000 candidates
idx, d = brute_force_knn(Q, X, k=5)
print(idx.shape)  # (10, 5)
Section 4

Locality-Sensitive Hashing (LSH)

LSH hashes vectors so that nearby vectors collide with high probability and far-apart vectors collide with low probability. Random projection LSH draws $b$ random hyperplanes and assigns each vector a $b$-bit bucket code. Only vectors sharing a bucket are compared exactly.

$$h(\mathbf{x})=\operatorname{sign}(\mathbf{r}\cdot\mathbf{x}),\quad \mathbf{r}\sim\mathcal{N}(0,\mathbf{I})$$ $$P[h(\mathbf{u})=h(\mathbf{v})]=1-\frac{\theta(\mathbf{u},\mathbf{v})}{\pi}$$ 
import torch

class RandomProjectionLSH:
    def __init__(self, d, b):
        """d: vector dim, b: number of hash bits"""
        self.R = torch.randn(b, d)   # b random hyperplanes

    def hash(self, X):
        """X: (N,d) -> (N,b) binary codes"""
        proj = X @ self.R.T          # (N,b)
        return (proj > 0).int()      # binary

    def bucket_id(self, X):
        bits = self.hash(X)          # (N,b)
        powers = 2 ** torch.arange(bits.size(1))
        return (bits * powers).sum(-1)  # scalar bucket id

lsh = RandomProjectionLSH(d=128, b=16)
X = torch.randn(5000, 128)
buckets = lsh.bucket_id(X)
print(buckets.shape)  # (5000,)
Section 5

Product Quantization (PQ)

PQ compresses a $d$-dimensional vector into $m$ short integers by splitting it into $m$ sub-vectors of dimension $d/m$ and quantizing each sub-vector against a small codebook of $k$ centroids learned by k-means. Distances are approximated using precomputed lookup tables, enabling billion-scale search with a few bytes per vector.

$$d(\mathbf{q},\mathbf{x})\approx\sum_{j=1}^{m}d\!\left(\mathbf{q}_j,\,\mathbf{c}_{j,\,\text{code}_j}\right)$$

Storage: $m\cdot\lceil\log_2 k\rceil$ bits vs $d\cdot 32$ bits (e.g. $m=8,k=256$: 8 bytes vs 3072 bytes for $d=768$).

import torch

def fit_pq(X, m, k, iters=20):
    """X: (N,d). Returns codebooks list of length m."""
    N, d = X.shape
    sub = d // m
    codebooks = []
    for j in range(m):
        Xj = X[:, j*sub:(j+1)*sub]         # (N, sub)
        # random init centroids
        idx = torch.randperm(N)[:k]
        C = Xj[idx].clone()                 # (k, sub)
        for _ in range(iters):
            dists = torch.cdist(Xj, C)      # (N, k)
            assign = dists.argmin(-1)       # (N,)
            for c in range(k):
                mask = assign == c
                if mask.any():
                    C[c] = Xj[mask].mean(0)
        codebooks.append(C)
    return codebooks

X = torch.randn(2000, 64)
cb = fit_pq(X, m=8, k=256)
print(len(cb), cb[0].shape)  # 8, (256,8)
Section 6

HNSW Graph Index

Hierarchical Navigable Small World (HNSW) builds a multi-layer proximity graph. Upper layers are sparse long-range skip graphs; the bottom layer contains all nodes. A query descends greedily from the top, entering the next layer at the current best candidate, achieving $O(\log N)$ search complexity with very high recall.

Layer $l$ contains $\approx N\cdot e^{-l\,m_L}$ nodes. Query complexity: $O(\log N)$.

import faiss, numpy as np

d, M = 128, 32          # dimension, HNSW connectivity
index = faiss.IndexHNSWFlat(d, M)
index.hnsw.efConstruction = 200

# Add vectors
X = np.random.randn(100_000, d).astype(np.float32)
index.add(X)

# Search
q = np.random.randn(10, d).astype(np.float32)
index.hnsw.efSearch = 64
D, I = index.search(q, k=5)
print(I.shape)  # (10, 5)
Section 7

Bi-Encoder vs Cross-Encoder

Bi-encoders encode query and document independently, allowing document embeddings to be pre-indexed. Cross-encoders process the concatenated pair, enabling full attention interaction for higher quality — but at $O(N)$ inference cost. Production systems typically use a bi-encoder for recall and a cross-encoder for re-ranking.

$$\text{bi-enc}:\quad s(\mathbf{q},\mathbf{d})=\text{enc}(\mathbf{q})\cdot\text{enc}(\mathbf{d}) \qquad O(1)\text{ retrieval}$$ $$\text{cross-enc}:\quad s(\mathbf{q},\mathbf{d})=\text{enc}([\mathbf{q};\mathbf{d}]) \qquad O(N)\text{ cost}$$
Model Retrieval Speed Quality Use Case
Bi-encoderO(1) w/ indexGoodFirst-stage recall
Cross-encoderO(N) per queryBestRe-ranking top-K
 
import torch, torch.nn.functional as F
from transformers import AutoModel, AutoTokenizer

tok = AutoTokenizer.from_pretrained("sentence-transformers/all-MiniLM-L6-v2")
enc = AutoModel.from_pretrained("sentence-transformers/all-MiniLM-L6-v2")

def mean_pool(out, mask):
    t = out.last_hidden_state * mask.unsqueeze(-1)
    return t.sum(1) / mask.sum(1, keepdim=True)

def embed(texts):
    batch = tok(texts, return_tensors="pt",
                padding=True, truncation=True)
    with torch.no_grad():
        out = enc(**batch)
    return F.normalize(mean_pool(out, batch["attention_mask"]), dim=-1)

q_emb = embed(["What is RAG?"])
d_emb = embed(["Retrieval-Augmented Generation combines LLMs with search."])
print((q_emb @ d_emb.T).item())  # cosine score
Section 8

Dense Passage Retrieval (DPR)

DPR trains separate query and passage BERT encoders with a contrastive loss. Hard negatives mined from BM25 top results (that are not the positive passage) dramatically improve retrieval quality over random negatives. In-batch negatives recycle the $B-1$ other batch positives as cheap negatives with no extra forward passes.

$$\mathcal{L}_{DPR}=-\log\frac{e^{\mathbf{q}^\top\mathbf{p}^+}}{e^{\mathbf{q}^\top\mathbf{p}^+}+\displaystyle\sum_{j}e^{\mathbf{q}^\top\mathbf{p}_j^-}}$$ 
import torch, torch.nn.functional as F

def dpr_in_batch_loss(q_enc, p_enc, tau=1.0):
    """
    q_enc: (B, d) query embeddings (normalized)
    p_enc: (B, d) positive passage embeddings (normalized)
    In-batch: passage i is negative for all queries j != i
    """
    q = F.normalize(q_enc, dim=-1)
    p = F.normalize(p_enc, dim=-1)

    scores = q @ p.T / tau    # (B, B)
    # diagonal = positive pairs
    labels = torch.arange(scores.size(0), device=scores.device)
    loss = F.cross_entropy(scores, labels)
    # Accuracy: fraction of queries retrieving correct passage
    acc = (scores.argmax(-1) == labels).float().mean()
    return loss, acc

B, d = 32, 768
loss, acc = dpr_in_batch_loss(torch.randn(B, d), torch.randn(B, d))
print(f"loss={loss:.3f}  acc={acc:.3f}")
Section 9

RAG: Retrieval-Augmented Generation Math

RAG marginalizes over the top-$k$ retrieved documents $z$ when generating output $y$ given input $x$. The retriever $p_\eta(z|x)$ scores documents; the generator $p_\theta(y|x,z)$ conditions on the retrieved context. RAG-Sequence retrieves once per query; RAG-Token retrieves at each generation step, yielding richer grounding at higher compute cost.

$$P_{\text{RAG}}(y|x)=\sum_{z\in\text{top-}k(p_\eta(z|x))}p_\eta(z|x)\cdot p_\theta(y|x,z)$$ 
import torch, torch.nn.functional as F

def rag_score(query_emb, doc_embs, doc_log_probs,
              lm_log_probs_per_doc):
    """
    query_emb      : (d,)
    doc_embs       : (k, d)     retrieved doc embeddings
    doc_log_probs  : (k,)       log p_eta(z|x) from retriever
    lm_log_probs   : (k,)       log p_theta(y|x,z) from LM
    Returns marginalised log-probability.
    """
    # Normalize retriever scores over top-k
    ret_log = F.log_softmax(doc_log_probs, dim=0)   # (k,)
    # RAG-Sequence: sum p(z|x)*p(y|x,z)
    joint = ret_log + lm_log_probs_per_doc           # (k,)
    marginal = torch.logsumexp(joint, dim=0)         # scalar
    return marginal

k, d = 5, 128
score = rag_score(torch.randn(d),
                  torch.randn(k, d),
                  torch.randn(k),
                  torch.randn(k))
print(f"log P_RAG(y|x) = {score:.3f}")

Embedding Dimensionality & Matryoshka

Matryoshka Representation Learning (MRL) trains a single model so any prefix $\mathbf{e}_{1:d'}$ of the full embedding is itself a strong embedding at dimension $d'$. This enables flexible storage–recall trade-offs at inference: use $d'=64$ for fast first-stage retrieval and $d'=768$ for re-ranking, all from one model.

$$\mathcal{L}_{\text{MRL}}=\sum_{d'\in M}\frac{1}{|M|}\mathcal{L}\!\left(f_{d'}(\mathbf{x})\right), \quad M=\{8,16,32,\ldots,d\}$$

PCA can further reduce storage: project $\mathbf{e}\in\mathbb{R}^d$ to top-$d'$ principal components, trading recall for compression.

import torch, torch.nn.functional as F

def mrl_loss(full_emb, labels, dims=(8,16,32,64,128)):
    """
    full_emb : (B, D) full-dimension embeddings
    labels   : (B,)   class labels for classification loss
    dims     : tuple  Matryoshka nesting dimensions
    """
    total = 0.0
    for d in dims:
        sub = full_emb[:, :d]                         # prefix slice
        sub = F.normalize(sub, dim=-1)
        # Simple linear head (in practice share across dims)
        logits = sub @ sub.T                          # (B,B) sim matrix
        total += F.cross_entropy(logits, labels)
    return total / len(dims)

B, D = 32, 128
emb = torch.randn(B, D, requires_grad=True)
lbl = torch.randint(0, B, (B,))
loss = mrl_loss(emb, lbl)
print(f"MRL loss = {loss.item():.3f}")
