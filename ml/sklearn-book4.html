Saving & Loading Pipelines

joblib · pickle · versioning · reproducible artefacts

🔑 Golden Rule

Always serialise the complete Pipeline — not just the final estimator. This guarantees preprocessing (scaler, encoder, imputer) is applied identically at serving time, eliminating train/serve skew — the #1 source of silent model degradation in production.

python — model versioning and artefact management
import joblib, json, hashlib
from datetime import datetime
from pathlib import Path

def save_model_artefact(pipeline, metrics, version):
    artefact_dir = Path(f"models/v{version}")
    artefact_dir.mkdir(parents=True, exist_ok=True)

    # Save pipeline
    model_path = artefact_dir / "pipeline.pkl"
    joblib.dump(pipeline, model_path, compress=3)

    # Compute checksum
    checksum = hashlib.md5(model_path.read_bytes()).hexdigest()

    # Save metadata sidecar
    meta = {
        "version":   version,
        "created_at": datetime.utcnow().isoformat(),
        "metrics":   metrics,
        "checksum":  checksum,
        "sklearn_version": __import__('sklearn').__version__,
    }
    (artefact_dir / "metadata.json").write_text(json.dumps(meta, indent=2))
    return str(model_path)

path = save_model_artefact(
    pipeline=full_pipeline,
    metrics={"auc": 0.934, "f1": 0.82},
    version="2024.01.15"
)

from fastapi import FastAPI
from pydantic import BaseModel
import joblib, pandas as pd, numpy as np
from typing import Optional

app = FastAPI(title="Loan Scoring API")
pipeline = joblib.load("models/v2024.01.15/pipeline.pkl")

class LoanApplication(BaseModel):
    age: int
    income: float
    loan_amount: float
    employment_years: float
    credit_history_months: int
    num_open_accounts: int
    debt_to_income: float
    purpose: str   # "home", "car", "education", "personal"

class ScoreResponse(BaseModel):
    probability_default: float
    decision: str   # "APPROVE" | "MANUAL_REVIEW" | "DECLINE"
    risk_band: str  # "LOW" | "MEDIUM" | "HIGH"

@app.post("/score", response_model=ScoreResponse)
async def score_application(app_data: LoanApplication):
    df = pd.DataFrame([app_data.dict()])
    prob = float(pipeline.predict_proba(df)[0, 1])

    if   prob < 0.1:  decision, band = "APPROVE",        "LOW"
    elif prob < 0.3:  decision, band = "MANUAL_REVIEW",  "MEDIUM"
    else:             decision, band = "DECLINE",         "HIGH"

    return ScoreResponse(probability_default=round(prob,4), decision=decision, risk_band=band)

@app.get("/health")
async def health(): return {"status": "ok"}

Data & Concept Drift Detection

PSI · KS test · feature distribution monitoring · silent failures

⚠️ The Silent Killer

A model can have 94% AUC in evaluation and silently degrade to 71% in production six months later — without any code change. Data drift (input distribution shift) and concept drift (the relationship between features and target changes) are the two failure modes. You must monitor for both.

python — PSI and KS-test drift detection
import numpy as np
from scipy.stats import ks_2samp

def population_stability_index(expected, actual, n_bins=10):
    """PSI < 0.1: stable. 0.1-0.2: minor shift. >0.2: major drift."""
    bins = np.percentile(expected, np.linspace(0, 100, n_bins+1))
    bins[0] -= 1e-9; bins[-1] += 1e-9
    exp_pct = np.histogram(expected, bins=bins)[0] / len(expected)
    act_pct = np.histogram(actual,   bins=bins)[0] / len(actual)
    exp_pct = np.clip(exp_pct, 1e-9, None)
    act_pct = np.clip(act_pct, 1e-9, None)
    psi = np.sum((act_pct - exp_pct) * np.log(act_pct / exp_pct))
    return psi

# Monitor every feature daily
drift_report = {}
for feat in numeric_features:
    psi  = population_stability_index(X_train[feat], X_today[feat])
    ks   = ks_2samp(X_train[feat], X_today[feat])
    drift_report[feat] = {
        "psi": round(psi, 4),
        "ks_stat": round(ks.statistic, 4),
        "ks_pvalue": round(ks.pvalue, 4),
        "drift_flag": psi > 0.2 or ks.pvalue < 0.01,
    }

drifted = [f for f, d in drift_report.items() if d["drift_flag"]]
print(f"Drifted features: {drifted}")

Production Monitoring

Score distribution · prediction logging · performance degradation alerts

python — rolling performance monitoring with shadow labels
import pandas as pd
from sklearn.metrics import roc_auc_score
from datetime import timedelta

# Predictions log: prediction_date, probability, true_label (arrives with delay)
log = pd.read_parquet("prediction_log.parquet")
log = log.dropna(subset=["true_label"])   # keep only resolved labels

# Rolling 7-day AUC window
log["prediction_date"] = pd.to_datetime(log["prediction_date"])
log = log.sort_values("prediction_date")

results = []
window = timedelta(days=7)
dates = pd.date_range(log["prediction_date"].min() + window,
                        log["prediction_date"].max(), freq="1D")

for end_date in dates:
    window_data = log[log["prediction_date"].between(end_date - window, end_date)]
    if len(window_data) < 100: continue
    auc = roc_auc_score(window_data["true_label"], window_data["probability"])
    results.append({"date": end_date, "rolling_auc": auc})

perf_df = pd.DataFrame(results)
# Alert if AUC drops > 0.03 below baseline
baseline_auc = 0.934
alerts = perf_df[perf_df["rolling_auc"] < baseline_auc - 0.03]
print(f"Performance alerts: {len(alerts)} windows below threshold")

Automated Retraining

Trigger policies · champion-challenger · shadow deployment

⏰

Time-Based Retraining

Simple schedule — retrain weekly/monthly regardless of drift signals.

  • Predictable compute budget
  • Best for stable distributions (e.g. logistics pricing)
  • Risk: retrain even when not needed (wasted compute)
📡

Trigger-Based Retraining

Retrain when drift PSI > threshold or rolling AUC drops.

  • More reactive, less wasteful
  • Requires monitoring infrastructure
  • Best for volatile distributions (e.g. fraud, markets)
🥊

Champion-Challenger

New model gets 10% traffic; promote only if it beats champion.

  • Zero-risk production deployment
  • Statistical significance testing before promotion
  • Requires A/B traffic splitting infrastructure
python — automated retraining pipeline with model promotion
from sklearn.model_selection import cross_val_score
from sklearn.metrics import roc_auc_score
import joblib, json
from pathlib import Path

def retrain_and_evaluate(X_new, y_new, current_model_path, candidate_pipeline):
    # Train candidate on fresh data
    candidate_pipeline.fit(X_new, y_new)
    candidate_auc = cross_val_score(
        candidate_pipeline, X_new, y_new, cv=5, scoring='roc_auc').mean()

    # Load champion AUC from metadata
    meta_path = Path(current_model_path).parent / "metadata.json"
    champion_auc = json.loads(meta_path.read_text())["metrics"]["auc"]

    print(f"Champion AUC: {champion_auc:.4f} | Candidate AUC: {candidate_auc:.4f}")

    if candidate_auc > champion_auc + 0.005:   # minimum meaningful improvement
        new_version = save_model_artefact(candidate_pipeline,
            {"auc": round(candidate_auc, 4)}, version="new")
        print(f"✅ Candidate promoted → {new_version}")
        return True
    print("⏸️  Champion retained — candidate did not improve")
    return False

Explainability with SHAP

Global feature importance · individual prediction explanation · regulatory compliance

⚖️ Real-World — EU AI Act Compliance for Credit Decisions

EU regulations require lenders to explain automated credit decisions. SHAP values provide per-application explanations: "Application declined due to: high debt-to-income ratio (+0.34), no credit history (+0.21), recently opened accounts (+0.11)." This satisfies the "right to explanation" requirement.

python — SHAP explanations for credit decisions

import numpy as np

explainer = shap.TreeExplainer(model)

# Global feature importance
shap.summary_plot(shap_values, preprocessed_X_test, feature_names=feature_names)

# Single prediction explanation (API response)
def explain_decision(application_idx, top_n=3):
    sv = shap_values[application_idx]
    pairs = sorted(zip(feature_names, sv), key=lambda x: abs(x[1]), reverse=True)[:top_n]
    reasons = []
    for feat, impact in pairs:
        direction = "increases" if impact > 0 else "decreases"
        reasons.append(f"{feat} {direction} default risk by {abs(impact):.3f}")
    return reasons
📌 Key Takeaways from Book 4
  • Serialise the full Pipeline — always. Never just the model.
  • Monitor PSI and KS-test on input features daily — catch drift before it hurts performance
  • Champion-challenger deployment is the gold standard for risk-free model updates
  • SHAP tree explainer is fast enough for real-time explanation APIs
  • Rolling performance windows are more robust than point-in-time evaluation for detecting degradation
